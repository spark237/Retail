name: Frontend CI

# Required status check name for branch protection: "Frontend CI / build-and-test"
# (see docs/BRANCH_PROTECTION.md)

on:
  push:
    branches: [main]
    paths:
      - 'retailguard-pwa/**'
      - '.github/workflows/frontend-ci.yml'
  pull_request:
    branches: [main]
    paths:
      - 'retailguard-pwa/**'
      - '.github/workflows/frontend-ci.yml'

defaults:
  run:
    working-directory: retailguard-pwa

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Set up Node
        uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: 'npm'
          cache-dependency-path: retailguard-pwa/package-lock.json

      - name: Install dependencies
        run: npm ci

      # Lint currently fails CI only on errors, not warnings.
      # 24 pre-existing warnings (unused imports/vars) are tracked as a
      # Sprint 0/1 cleanup item, not a merge blocker. oxlint exits non-zero
      # on errors regardless, so this step still gates real problems.
      - name: Lint (oxlint)
        run: npm run lint

      - name: Run tests
        run: npm run test

      - name: Run tests with coverage
        run: npm run test:coverage

      - name: Upload coverage report
        uses: actions/upload-artifact@v4
        with:
          name: frontend-coverage
          path: retailguard-pwa/coverage/
        # Informational only for now — no frontend coverage threshold has
        # been set (see Phase 4 Baseline Plan coverage table, which only
        # defines backend targets). This step does not fail the build.

      - name: Build (includes PWA / service worker generation)
        run: npm run build
